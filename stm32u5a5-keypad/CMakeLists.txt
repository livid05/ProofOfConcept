cmake_minimum_required(VERSION 3.22)
set(CMAKE_SYSTEM_NAME Generic)
set(CMAKE_SYSTEM_PROCESSOR arm)
set(CMAKE_TRY_COMPILE_TARGET_TYPE STATIC_LIBRARY)
project(Keypad_U5A5 C ASM)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(HAL firmware/Drivers/STM32U5xx_HAL_Driver)
set(HAL_NAMES hal hal_cortex hal_dma hal_dma_ex hal_exti hal_flash hal_flash_ex hal_gpio hal_pwr hal_pwr_ex hal_rcc hal_rcc_ex hal_pcd hal_pcd_ex ll_usb)
foreach(name IN LISTS HAL_NAMES)
 list(APPEND HAL_SOURCES ${HAL}/Src/stm32u5xx_${name}.c)
endforeach()
file(GLOB APP_SOURCES CONFIGURE_DEPENDS firmware/Core/Src/*.c)
file(GLOB USB_SOURCES firmware/USB/Core/Src/usbd_*.c)
list(FILTER USB_SOURCES EXCLUDE REGEX "template")
file(GLOB STARTUP firmware/Core/Startup/*.s)
add_executable(keypad ${APP_SOURCES} ${USB_SOURCES} ${HAL_SOURCES} ${STARTUP})
set_target_properties(keypad PROPERTIES SUFFIX .elf)
target_include_directories(keypad PRIVATE firmware/Core/Inc ${HAL}/Inc ${HAL}/Inc/Legacy firmware/Drivers/CMSIS/Include firmware/Drivers/CMSIS/Device/ST/STM32U5xx/Include firmware/USB/Core/Inc)
target_compile_definitions(keypad PRIVATE USE_HAL_DRIVER STM32U5A5xx)
target_compile_options(keypad PRIVATE -mcpu=cortex-m33 -mthumb -mfpu=fpv5-sp-d16 -mfloat-abi=hard -g3 -Og -ffunction-sections -fdata-sections -Wall -Wextra -Werror)
target_link_options(keypad PRIVATE -mcpu=cortex-m33 -mthumb -mfpu=fpv5-sp-d16 -mfloat-abi=hard "-T${CMAKE_SOURCE_DIR}/firmware/STM32U5A5RJTX_FLASH.ld" --specs=nano.specs --specs=nosys.specs -Wl,--gc-sections,-Map=keypad.map -Wl,--print-memory-usage)
add_custom_command(TARGET keypad POST_BUILD COMMAND ${CMAKE_OBJCOPY} -O binary $<TARGET_FILE:keypad> ${CMAKE_BINARY_DIR}/keypad.bin COMMAND ${CMAKE_OBJCOPY} -O ihex $<TARGET_FILE:keypad> ${CMAKE_BINARY_DIR}/keypad.hex COMMAND ${CMAKE_SIZE} $<TARGET_FILE:keypad>)
